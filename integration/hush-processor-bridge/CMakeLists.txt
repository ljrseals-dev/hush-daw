cmake_minimum_required(VERSION 3.20)
project(HushProcessorBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(hush_processor_bridge STATIC src/ProcessorBridge.cpp)
target_include_directories(hush_processor_bridge PUBLIC include)
target_compile_features(hush_processor_bridge PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(hush_processor_bridge PRIVATE /W4 /permissive-)
else()
  target_compile_options(hush_processor_bridge PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
enable_testing()
add_executable(hush_bridge_smoke tests/bridge_smoke.cpp)
target_link_libraries(hush_bridge_smoke PRIVATE hush_processor_bridge)
add_test(NAME hush_bridge_smoke COMMAND hush_bridge_smoke)
